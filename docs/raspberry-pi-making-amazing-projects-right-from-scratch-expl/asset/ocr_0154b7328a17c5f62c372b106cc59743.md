def setHome(ser):
    for i in range(0, 9):
        setAngle(ser, i, 90)

    setAngle(ser, 5, 70)
    setAngle(ser, 0, 80)
    time.sleep(.5)
    setAngle(ser, 5, 90)
    setAngle(ser, 7, 110)
    setAngle(ser, 8, 80)

ser = serial.Serial("/dev/ttyACM0", 9600)

setHome(ser)
time.sleep(2)
liftLeftLeg(ser)
time.sleep(.5)

while 1:
    servo = int(raw_input("Servo number: "))
