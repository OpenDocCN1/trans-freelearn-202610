function showContent(element)
{
    var value=element.text.replace(' ','');
    NavigationTree.getContent(value,{
        callback:function(content)
        {
            var treeMenuContent=dwr.util.byId('treeMenuContent');
            treeMenuContent.innerHTML=content;
        }
    });
}
