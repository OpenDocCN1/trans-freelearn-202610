    ServiceReference1.WcfServiceOf_PayMessage_ErrorCodesClient client1 =
        new ServiceReference1.WcfServiceOf_PayMessage_ErrorCodesClient();
    ErrorCodes returnCode = client1.Process(message);
    user.errorCode = returnCode;
    return View(user);
}
