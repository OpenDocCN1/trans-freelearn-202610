(use '[clojure.java.shell])
; set this to 0 (off) or 1 (on)
(defn set-led [n]
  (sh "sh" "-c"
  (str "echo " n " > /sys/class/leds/led0/brightness")))

; now some looping
(dotimes [i 10] (set-led 0) (Thread/sleep 1000) (set-led 1))
