```
def imshow(image):
    dpi = plt.rcParams["figure.dpi"]
    im_data = image[:, :, :-1]
    height, width, depth = im_data.shape
    figsize = width / float(dpi), height / float(dpi)
    fig = plt.figure(figsize=figsize)
    plt.imshow(im_data)
    plt.imshow(im_data)
    plt.axis("off")
    plt.show()
```

```
def visualize_output(img, output, cfg, scale=1.0):
```
