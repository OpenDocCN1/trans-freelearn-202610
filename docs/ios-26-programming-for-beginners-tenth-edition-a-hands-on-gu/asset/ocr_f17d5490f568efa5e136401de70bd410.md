```
private var annotations: [JournalEntry] = []
```

This property will store the JournalEntry instance for the MKAnnotationView instance that was tapped.

5. Add mapView(_:annotationView:calloutAccessoryControlTapped:) after the mapView(_:viewFor:) method:

```
func mapView(_ mapView: MKMapView, annotationView view: MKAnnotationView, calloutAccessoryControlTapped control: UIControl) {
    guard let annotation = mapView.selectedAnnotations.first as? JournalEntry else {
        return
    }
    selectedAnnotation = annotation
    performSegue(withIdentifier: "showMapDetail", sender: self)
}
```
