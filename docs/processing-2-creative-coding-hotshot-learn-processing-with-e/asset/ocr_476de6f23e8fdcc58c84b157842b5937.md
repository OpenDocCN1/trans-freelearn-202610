void setup() {
    size(300,300);

    minim = new Minim( this );
    player = minim.loadFile( "loop.mp3", 1024 );

    visualizers = new Visualizer[] {
        new BarHorizontal(),
        new Alternate()
    };

    viz = visualizers[0];
    viz.setActive( true );
    Thread t = new Thread() {
        public void run() {
            for(;;) {
                viz.tick();
                delay(125);
            }
        }
    };
    t.start();

    Thread t2 = new Thread() {
        public void run() {
            for(;;) {
                viz.setActive( false );
                viz = visualizers[ int(random
                ( visualizers.length))];
                viz.init();
                viz.setActive( true );
                delay(8000);
            }
        }
    };
    t2.start();
    player.loop();
}
