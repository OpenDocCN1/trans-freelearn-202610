g.background(0);
float w = 320/8;
for( int x = 0; x < 8; x++ ) {
for( int y = 0; y < 8; y++ ) {
g.stroke( 255 );

if ( viz.tiles[x][y] > 32 )
{ viz.tiles[x][y] -= ( viz.tiles[x][y] -32)/25; }
g.rect( w * x, w*y, w, w );
}
}
g.endDraw();
}
