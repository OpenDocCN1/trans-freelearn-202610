void draw() {
  background(0);
  float w = 300/8;
  for ( int x = 0; x < 8; x++ ) {
    for ( int y = 0; y < 8; y++ ) {
      stroke( 255 ) ;
      fill( viz.tiles[ x ][ y ],0,0 ) ;
      if ( viz.tiles[ x ][ y ] > 32 ) {
        viz.tiles[ x ][ y ] -= (viz.tiles[ x ][ y ]-32)/25 ;
      }
      rect( w * x, w*y, w, w ) ;
    }
  }
}
