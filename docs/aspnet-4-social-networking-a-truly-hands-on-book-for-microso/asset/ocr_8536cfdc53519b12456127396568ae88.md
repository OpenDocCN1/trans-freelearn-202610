public void DeleteProfile(Profile profile)
{
    using (FisharooDataContext dc = conn.GetContext())
    {
        dc.Profiles.DeleteObject(dc.Profiles
            .Where(pf => pf.ProfileID
            .Equals(profile.ProfileID))
            .FirstOrDefault());
        dc.SaveChanges();
    }
}
