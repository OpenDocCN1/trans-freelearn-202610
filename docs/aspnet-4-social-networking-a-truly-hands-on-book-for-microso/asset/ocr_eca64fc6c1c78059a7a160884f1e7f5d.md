    //flag marked as friends only test
    else if (Flags.Where(f => f.PrivacyFlagTypeID ==
            PrivacyFlagTypeID && f.VisibilityLevelID ==
            (int)VisibilityLevel.VisibilityLevels
            Friends).
        FirstOrDefault() != null && isFriend)
        result = true;
    else if (Flags.Where(f => f.PrivacyFlagTypeID ==
