```
// GET: Posts/Delete/5
public async Task<IActionResult> Delete(int? id)
{
    if (id == null)
    {
        return NotFound();
    }

    var post = await _repositoryWithQueries.GetSingleAsync(_context,
        new GetPostByIdQuery(id, true));
    if (post == null)
    {
        return NotFound();
    }

    return View(post);
}
```
