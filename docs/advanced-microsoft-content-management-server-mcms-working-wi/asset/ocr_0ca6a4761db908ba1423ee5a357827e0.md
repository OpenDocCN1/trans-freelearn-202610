7. As a convenience in development, we will create the BoundXmlPlaceholder property on the class. This property casts the inherited BoundPlaceholder property to an XmlPlaceholder:

```
#region Properties

/// <summary>
/// Returns the Strongly Typed
/// Bound XmlPlaceholder
/// </summary>
private XmlPlaceholder BoundXmlPlaceholder
{
    get
    {
        return (this.BoundPlaceholder as XmlPlaceholder);
    }
}

#endregion
```
