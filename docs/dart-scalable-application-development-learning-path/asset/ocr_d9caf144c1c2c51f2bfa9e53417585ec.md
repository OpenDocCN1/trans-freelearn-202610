String getRandomTile() {
    var num = rand.nextInt(10);
    var imgUrl = tiles[num];
    while (imgUrl == null) {
        num = rand.nextInt(10);
        imgUrl = tiles[num];
    }
    tiles.remove(num);
    return imgUrl;
}
