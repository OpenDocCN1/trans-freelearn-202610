Future<String> getItem(key) {
  return new Future.sync(() {
    return window.localStorage[key];
  });
}

Future setItem(key, value) {
  return new Future.sync(() {
    window.localStorage[key] = value;
  });
}
Future removeItem(key) {
  return new Future.sync(() {
    window.localStorage.remove(key);
  });
}
