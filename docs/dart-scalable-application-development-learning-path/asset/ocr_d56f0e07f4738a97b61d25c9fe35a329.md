openAndRead() {
    connect(uri).then((_conn) {
        conn = _conn;
        readData();
    })
    .catchError(print);
}

// 8- reading records from a table:
readData() {
    conn.query('select * from jobs').toList().then((results) {
        processJob(results);
        close();
    });
}

// 9- working with record data:

    for (var row in results) {
        // Refer to columns by nam:
