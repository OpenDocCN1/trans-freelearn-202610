Contracts can be compiled by using the command shown in the following screenshot. Note that it takes that contract file name as an argument. After successful compilation, all relevant JSON files will be written under ./meta directory.

```
drequinox@drequinox-OP7010:~/testApp$ bloc compile Greeter
Compiling single contract: Greeter.sol
Compile successful: contract mortal {
  /* Define variable owner of the type address*/
  address owner;

  /* this function is executed at initialization and sets the owner of the contract */
  function mortal() { owner = msg.sender; }

  /* Function to recover the funds on the contract */
  function kill() { if (msg.sender == owner) suicide(owner); }
}

contract Greeter is mortal {
  /* define variable greeting of the type string */
  string greeting;

  /* this runs when the contract is executed */
  function Greeter(string _greeting) public {
    greeting = _greeting;
  }

  /* main function */
  function greet() constant returns (string) {
    return greeting;
  }
}
```
