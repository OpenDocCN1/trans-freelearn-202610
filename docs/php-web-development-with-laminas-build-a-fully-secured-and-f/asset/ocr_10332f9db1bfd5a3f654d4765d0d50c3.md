Observe that index.phtml imports an index.js file. The index.js file is responsible for submitting HTTP requests to the login API. Let us see the source code of index.js:

```
var login = function() {
	var api = $("#api").attr("url");
	var target = $("#target").attr("url");
	var formData = $("#user").serialize();
	$.post(api, formData, function(data) {
		if (!data.logged) {
			alert('Invalid data!');
			return;
		}

		alert("user has been logged successfully");
		window.location.href = target;
	}).fail(function() {
		alert('error');
	});
```
