kvs.[value]
FROM kvs, key_tab,application_identifier_tab
WHERE key_tab.[id] = '1'
AND kvs.[key] = key_tab.[id]
AND application_identifier_tab.[id] = kvs.
[application_identifier]
ORDER BY application_identifier_tab.[id];
