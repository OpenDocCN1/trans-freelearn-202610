```
except ValueError:
    # not an int
    pass
# put a comma between the entities
if ecounter > 0:
    eoutput += ','
eoutput += json.dumps(erow)
    ecounter += 1
eoutput += ']'

# write out a new file to disk
f.write(',')
f.write('"links":')
f.write(eoutput)
f.write('}')
f.close()
```
