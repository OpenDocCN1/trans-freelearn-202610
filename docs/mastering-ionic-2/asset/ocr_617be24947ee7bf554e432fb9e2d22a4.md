The next important thing that we have to do is clearing old cache in case new versions of service workers create new `cacheStorage`. So we execute this during our activation of service workers:

```
```javascript
// Deleting old caches
self.addEventListener('activate', function(event) {
  var expectedCacheNames = Object.keys(self.toolbox.options.cache.name).map(function(key) {
    return self.toolbox.options.cache.name[key];
  });
  event.waitUntil(
    caches.keys().then(function(cacheNames) {
      return Promise.all(
        cacheNames.map(function(cacheName) {
          if (expectedCacheNames.indexOf(cacheName) === -1) {
            console.log('Deleting out of date cache:', cacheName);
            return caches.delete(cacheName);
          }
        })
      );
    })
  );
});
