def apply_color_pop(im):
    gray = rgb2gray(im)
    gray = np.stack((gray,)*3, axis=-1)
    hsv = color.rgb2hsv(im)
    mask_cond = (hsv[...,0] <= 5/180) | ((hsv[...,0] >= 175/180))
    gray[mask_cond] = im[mask_cond]
    return gray

im = imread('images/bus.jpg')
