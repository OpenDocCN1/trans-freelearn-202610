```
def CHSH_circuit(x,y):
    a0=0
    a1=np.pi/2
    b0=np.pi/4
    b1=-np.pi/4
    circ = QuantumCircuit(2,2)
    circ.h(0)
    circ.cx(0,1)
    if(x==0):
        circ.ry(a0,0)
    else:
        circ.ry(a1,0)
    if(y==0):
        circ.ry(b0,1)
    else:
        circ.ry(b1,1)
    circ.measure([0,1], [0,1])
    return circ
```
