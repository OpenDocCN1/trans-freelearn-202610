subscribeToNewLikes() {
    const onPostLikedObs = this.onPostLiked
        .subscribe();
    onPostLikedObs.subscribe({
        next: (result) => {
            const like = result.data?.onPostLiked;
            if (this.authUser?.id !== like?.user.id) {
                this.notificationsCount++;
                this.changeDetectorRef.markForCheck();
            }
        }
    });
}
