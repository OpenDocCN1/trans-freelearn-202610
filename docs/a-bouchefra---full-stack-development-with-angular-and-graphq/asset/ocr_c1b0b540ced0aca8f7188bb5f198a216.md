7. Define the following private methods to check whether a token exists in the local storage and whether it's expired or not:

```
```ts
private tokenExists(): boolean {
  return !!localStorage.getItem(ACCESS_TOKEN);
}
private tokenExpired(token: string): boolean {
  const tokenObj =
    JSON.parse(atob(token.split('.')[1]));
  return Date.now() > (tokenObj.exp * 1000);
}

```
```ts
private updateAuthState(token: string, user: User) {
  this.storeToken(token);
  this.storeUser(user);
  this.authSubject.next({
    isLoggedIn: true,
    currentUser: user,
    accessToken: token
  });
}
private resetAuthState() {
  this.authSubject.next({
    isLoggedIn: false,
    currentUser: null,
    accessToken: null
  });
}
