[MapperKind.OBJECT_FIELD]: (fieldConfig) => {
  const directiveName = 'auth';
  const authDirective = getDirective(schema,
    fieldConfig, directiveName)?.[0];
  if (authDirective) {
    const { resolve = defaultFieldResolver } =
      fieldConfig;
    fieldConfig.resolve = (source, args, context,
      info) => {
      if (!context.authUser) {
        throw new ApolloError("User is not
          authenticated", "USER_NOT_AUTHENTICATED");
      }
      return resolve(source, args, context, info);
    };
    return fieldConfig;
  }
}
