subscribeToNewComments() {
  const onPostCommentedObs = this.onPostCommentedGQL
    .subscribe();
  onPostCommentedObs
    .subscribe({
      next: (result) => {
        const comment = result.data?.onPostCommented;
        if (this.authUser?.id !== comment?.author.id) {
          this.notificationsCount++;
          this.changeDetectorRef.markForCheck();
        }
      }
    });
}

Then, we subscribed to the returned Observable to start listening for new comments.
